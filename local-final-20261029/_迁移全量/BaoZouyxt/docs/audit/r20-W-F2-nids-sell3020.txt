### 7) bz-rpc-server.js 中卖物品 handler 实现与 Effect 15 特判检查
- 路径与行号：F:\BaoZouyxt\hotfix-patch\core\bz-rpc-server.js:2194-2271
- 原文逐字实现：
```javascript
    saleItem: function (p) {
      p = p || {};
      var s = ensureLedger(loadSave());
      var arr = p.saleArr || [];
      var sold = [];
      var coins = 0, pricedN = 0, unpriced = [], ivch = {}, eqch = {};
      for (var i = 0; i < arr.length; i++) {
        var e = arr[i], id = 0, cnt = 1, equip = false;
        if (Array.isArray(e)) {
          // 客户端格式 [kind, id/uid, count?] ——
          //   kind: 1=物品(id是itemID)，2=装备(id是uid)
          //   见 tools/index.js:65191-65208（InventoryModel.sellItem）
          var kind = e[0];
          id = e[1];
          cnt = (e.length >= 3 && e[2] > 0) ? e[2] : 1;
          equip = (kind === 2);
        } else if (typeof e === 'object' && e) {
          id = e.id || e.itemID || e.uid;
          cnt = e.count || e.num || 1;
          equip = !!e.equip;
        } else {
          id = e; cnt = 1;
        }
        if (!id) continue;
        if (equip) {
          // 装备：id 是 uid 字符串；计价看 itemID；扣库存扣 playerEquip
          var haveE = (s.playerEquip && s.playerEquip[id] && s.playerEquip[id][1]) ? s.playerEquip[id][1] : 0;
          if (cnt > haveE) { env.log('saleItem 装备 ' + id + ' 截断：请求 x' + cnt + '，真实持有 ' + haveE); cnt = haveE; }
          if (cnt <= 0) continue;
          var itemID = (s.playerEquip[id] && s.playerEquip[id][0]) ? s.playerEquip[id][0] : id;
          var unitE = itemPrice(itemID);
          if (unitE > 0) { coins += unitE * cnt; pricedN += cnt; } else unpriced.push(itemID);
          s.playerEquip[id][1] -= cnt;
          if (s.playerEquip[id][1] <= 0) {
            delete s.playerEquip[id];
            eqch[id] = [0];
          } else {
            eqch[id] = [1, clone(s.playerEquip[id])];
          }
          sold.push('eq:' + id + '(item' + itemID + ')x' + cnt + '@' + unitE);
        } else {
          // 物品：id 是 itemID；计价看 itemPrice(id)；扣库存扣 bag
          var have = (s.bag && s.bag[id]) ? s.bag[id] : 0;
          if (cnt > have) { env.log('saleItem ' + id + ' 截断：请求 x' + cnt + '，真实库存只有 ' + have); cnt = have; }
          if (cnt <= 0) continue;
          var unit = itemPrice(id);
          if (unit > 0) { coins += unit * cnt; pricedN += cnt; } else unpriced.push(id);
          s.bag[id] -= cnt;
          if (s.bag[id] <= 0) delete s.bag[id];
          ivch[String(id)] = (s.bag[id] || 0);
          sold.push(id + 'x' + cnt + '@' + unit);
        }
      }
      var prevCoins = s.coins || 0;
      if (coins > 0) {
        s.coins = prevCoins + coins;
        // baseValueAy[35] 镜像同步（LoadGame / 存档对齐）
        if (Array.isArray(s.baseValueAy) && s.baseValueAy.length > 35) {
          s.baseValueAy[35] = s.coins;
        }
      }
      env.log('saleItem ' + sold.join(',')
        + ' → +铜钱 ' + coins + '（' + prevCoins + '→' + s.coins + '，计价x' + pricedN
        + (unpriced.length ? '，零价/无价品:' + unpriced.join('/') : '') + '）');
      writeSave(s);
      // R04 终验（2026-10-22）客户端消费端闭环：
      //   - ivch 必须填【绝对值】（0=卖光删格，>0=余量覆盖），index.js:64730 走 `a.default.saveAllItemCount(c, s[c])`；
      //   - eqch 必须填 `[0]`（卖光走 deleteEq）或 `[1, 数组]`（余量覆盖），index.js:64737；
      //   - eco 挂在 note 下触发 ECONOMY_CHNG（index.js:64700 直接赋值 RoleBaseInfo.coins）；
      //   - eRet 必须是 1（index.js:64727 检查 `1 == t.eRet`）。
      var payload = {
        eRet: 1,
        note: {
          ivch: ivch,
          eqch: eqch,
          eco: { coins: s.coins }
        }
      };
      // ⚠ note 必须落在 `User.saleItem` 里（消费端读 o.note.ivch），不能挂在信封顶层 ——
      //   shaped('saleItem', payload) 保证 User.saleItem.note 结构原样输出
      //   （若走旧 wrapped 会被打散）。同时 extra 注入 `{ achievement: ach() }`
      //   满足通用信封规范。
      return shaped('saleItem', payload) || wrapped('saleItem', payload, { achievement: ach() });
    },
```
